/** Recursive: a role growing a subdirectory must not escape the check. */
function sources(dir: string): string[] {
  const out: string[] = [];
  const walk = (d: string) => {
    for (const entry of readdirSync(d, { withFileTypes: true })) {
      const full = join(d, entry.name);
      if (entry.isDirectory()) walk(full);
      else if (entry.name.endsWith(".ts")) out.push(full);
    }
  };
  walk(join(SRC, dir));
  return out;
}

function importSpecifiers(text: string): string[] {

  for (const m of text.matchAll(re)) out.push(m[1]!);
  return out;
}

    assert.equal(decide(0n).allowed, false);
    assert.equal(decide(-1n).allowed, false);
  });

  test("the cap is configurable per instance", () => {
    assert.equal(decide(50_000n, { maxAtomicPerPayment: 100_000n }).allowed, true);

/**
 * Stipend's spend policy — the reason this project exists.
 *
 * Separate from the server because it is the only part a user of Stipend
 * would ever configure, and because it is the ONLY place in the system that
 * knows a refusal was a policy veto rather than a malformed payment. Both
 * reach the merchant as success:false, so that distinction has to be carried
 * deliberately rather than recovered from an error message later.
 *
 * Pure and synchronous today. When the Postgres ledger lands this becomes
 * async and takes a payment key, which is also what removes the single-flight
 * limitation below.
 */

export type PolicyDecision =
  | { allowed: true }
  | { allowed: false; reason: string };

export type PolicyConfig = {
  /** Per-payment cap in atomic token units. 10_000 = 0.01 USDC at 6 decimals. */
  maxAtomicPerPayment: bigint;
};

export const DEFAULT_POLICY: PolicyConfig = {
  maxAtomicPerPayment: 10_000n,
};

/**
 * Decide a single payment against the policy.
 *
 * Takes the amount already parsed, so this stays testable without a chain,
 * a server or a request.
 */
//! Understanding question - Why are we setting the config this way here .. looks like over engineered stuff?
export function decide(amountAtomic: bigint, config: PolicyConfig = DEFAULT_POLICY): PolicyDecision {
  if (amountAtomic <= 0n) {
    return { allowed: false, reason: `non-positive amount ${amountAtomic}` };
  }
  if (amountAtomic > config.maxAtomicPerPayment) {
    return {
      allowed: false,
      reason: `per-payment cap ${config.maxAtomicPerPayment} exceeded by ${amountAtomic}`,
    };
  }
  return { allowed: true };
}

/**
 * Settlement classification: did this payment reach the chain, and is it
 * safe to sign a replacement?
 *
 * The question that decides everything is "was a signature broadcast",
 * not "what did the error say". Error text is a hint. A signature is a fact.
 */

//! Use enums
export const SETTLEMENT_STATES = [
  "rejected",
  "expired",
  "pending",
  "confirmed",
  "failed",
] as const;

export type SettlementState = (typeof SETTLEMENT_STATES)[number];

export type Classification = {
  state: SettlementState;
  /** May the payer build and sign a NEW payment for this request, NOW? */
  resignable: boolean;
  /**
   * The refusal may lapse on its own: identical terms could succeed later
   * with nothing changed. A spend cap that resets, a rate limit, a
   * temporarily unreachable dependency.
   *
   * Distinct from `resignable`, which asks "is it safe right now". A
   * payment can be retryableLater and not resignable: wait, then retry.
   */
  retryableLater: boolean;
  /** Why this classification was reached. Goes in the ledger and the logs. */
  reason: string;
};

export type ClassifyInput = {
  /** Which half of the protocol produced this outcome. */
  phase: "verify" | "settle"; //! use enums
  /** Did that call report success? */
  ok: boolean;
  /** Transaction signature. Present ONLY once bytes were broadcast. */
  signature?: string | undefined;
  /** invalidMessage (verify) or errorReason (settle). Advisory only. */
  errorText?: string | undefined;
  /** From the 402: extra.lastValidBlockHeight. */
  lastValidBlockHeight: bigint;
  /** Chain height observed at classification time. */
  currentBlockHeight: bigint;
  /**
   * Why a refusal happened, supplied by whoever refused — NOT parsed from
   * errorText. The facilitator's policy hook knows it vetoed; the scheme
   * knows the transaction was malformed. Only they can say.
   *
   *   "policy"  — a limit or gate that may lapse (spend cap, rate limit)
   *   "terms"   — wrong by construction (bad amount, missing account)
   *   "unknown" — unattributed; treated as "terms", the safer default
   */
  refusalKind?: "policy" | "terms" | "unknown" | undefined; //! use enums
};

export function classify(input: ClassifyInput): Classification {
  const {
    phase,
    ok,
    signature,
    errorText,
    lastValidBlockHeight,
    currentBlockHeight,
    refusalKind,
  } = input;

  //! Refusal from whom would return the signature as empty?
  // SettleResponse.transaction is a REQUIRED string in @x402/core, so a
  // refusal that never broadcast arrives as "". Normalise here rather than
  // trusting every caller: an empty signature must never read as "broadcast".
  const sig = signature && signature.trim().length > 0 ? signature : undefined;

  const expiredByHeight = currentBlockHeight > lastValidBlockHeight;

  // ---- 1. Success, with proof. -------------------------------------------
  if (phase === "settle" && ok && sig) {
    return {
      state: "confirmed",
      resignable: false,
      retryableLater: false,
      reason: `settled: ${sig}`,
    };
  }

  // ---- 2. Success claimed with no signature. -----------------------------
  // A facilitator reporting success without a signature is misbehaving.
  // Treat it as broadcast-of-unknown-outcome rather than trusting it: the
  // expensive mistake is paying twice, not waiting.
  if (phase === "settle" && ok && !sig) {
    return {
      state: "pending",
      resignable: false,
      retryableLater: false,
      reason: "settle reported success with no signature; outcome unknown",
    };
  }

  // ---- 3. A signature exists. This dominates everything below. -----------
  // Bytes went to the chain. We cannot tell from here whether they landed:
  // the RPC may have failed while the transaction confirmed. The ONLY safe
  // move is to reconcile against the chain, never to re-sign.  //! How are we reconciling and where are we reconciling?
  //
  // Deliberately placed above the expiry check: an expired-looking error
  // AFTER broadcast does not make re-signing safe.
  if (sig) {
    return {
      state: "pending",
      resignable: false,
      retryableLater: false,
      reason: `broadcast ${sig}, outcome unknown${errorText ? `: ${errorText}` : ""}`,
    };
  }

  // ---- 4. Nothing was broadcast. Expired? --------------------------------
  // Structural test, not a string match. Past lastValidBlockHeight a Solana
  // transaction is permanently dead — it cannot land late. So the payer may
  // safely authorise a fresh one.
  if (expiredByHeight) {
    return {
      state: "expired",
      resignable: true,
      retryableLater: false,
      reason: `chain at ${currentBlockHeight} is past lastValidBlockHeight ${lastValidBlockHeight}; nothing broadcast`,
    };
  }

  // ---- 5. Nothing broadcast, still within the window. --------------------
  // The payment was refused on its merits: wrong amount, missing account,
  // policy veto. Re-signing identical terms produces an identical refusal,
  // so do not retry automatically — surface it.  //! What happens after it is surfaced? How are we handling post surfacing the scenario?
  // A policy refusal may lapse; a malformed payment never will. Unattributed
  // refusals are treated as terms — the safe default is "do not retry".
  return {
    state: "rejected",
    resignable: false,
    retryableLater: refusalKind === "policy",
    reason: `${phase} refused within the validity window (${refusalKind ?? "unknown"})${errorText ? `: ${errorText}` : ""}`,
  };
}

  mkdirSync(KEY_DIR, { recursive: true, mode: 0o700 });
  const path = `${KEY_DIR}/${name}.json`;

  let seed: Uint8Array;

  if (existsSync(path)) {
    const raw: unknown = JSON.parse(readFileSync(path, "utf8"));
    if (!Array.isArray(raw) || raw.length !== 32) {
      throw new Error(
        `${path}: expected a 32-element array, got ${JSON.stringify(raw).slice(0, 40)}`,
      );
    }
    seed = Uint8Array.from(raw as number[]);
  } else {
    seed = new Uint8Array(randomBytes(32));
    writeFileSync(path, JSON.stringify(Array.from(seed)), { mode: 0o600 });
  }

  return createKeyPairSignerFromPrivateKeyBytes(seed);
}

  TOKEN_PROGRAM_ADDRESS,
} from "@solana-program/token";
import { loadOrCreateSigner } from "../shared/keys";
import { createChain } from "../shared/chain";
import { loadConfig } from "./config";

const chain = createChain(loadConfig().chain);

const feePayer = await loadOrCreateSigner("fee-payer");
const merchant = await loadOrCreateSigner("merchant");

// Pure derivation: no network call. A token account's address is a function
// of (owner, mint, token program), so it is knowable before it exists.
const [ata] = await findAssociatedTokenPda({
  owner: merchant.address,
  mint: chain.usdcMint,
  tokenProgram: TOKEN_PROGRAM_ADDRESS,
});
console.log(`merchant ${merchant.address}`);
console.log(`its USDC account would be ${ata}`);

// payer and owner differ: the fee payer rents the account into existence,
// the merchant owns it. The merchant never needs SOL.
const createIx = await getCreateAssociatedTokenIdempotentInstructionAsync({
  payer: feePayer,
  owner: merchant.address,
  mint: chain.usdcMint,
});

const { value: latestBlockhash } = await chain.rpc.getLatestBlockhash().send();

// Four things every Solana transaction needs, in order: a version, a fee
// payer, a lifetime, and instructions. pipe() threads the message through
// each step — each returns a new message rather than mutating.
const message = pipe(
  createTransactionMessage({ version: 0 }),
  (m) => setTransactionMessageFeePayerSigner(feePayer, m),
  (m) => setTransactionMessageLifetimeUsingBlockhash(latestBlockhash, m),
  (m) => appendTransactionMessageInstruction(createIx, m),
);

const signed = await signTransactionMessageWithSigners(message);
assertIsTransactionWithBlockhashLifetime(signed);
